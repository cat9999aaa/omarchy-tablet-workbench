cmake_minimum_required(VERSION 3.20)
project(omarchy_touch_ime VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(Rime REQUIRED IMPORTED_TARGET rime)
find_package(Qt6 REQUIRED COMPONENTS Core DBus)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Fcitx5Core REQUIRED)
find_package(Fcitx5Module REQUIRED COMPONENTS DBus)
add_library(omt_engine STATIC src/engine.cpp)
target_link_libraries(omt_engine PUBLIC PkgConfig::Rime)
target_include_directories(omt_engine PUBLIC src)
add_executable(test_engine tests/test_engine.cpp)
target_link_libraries(test_engine PRIVATE omt_engine)
enable_testing()
add_test(NAME engine COMMAND test_engine ${CMAKE_CURRENT_SOURCE_DIR}/rime ${CMAKE_CURRENT_BINARY_DIR}/test-rime)
add_test(NAME backend COMMAND dbus-run-session -- ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_backend.py)
add_test(NAME control COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_control.py)
add_test(NAME layout COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_layout.py)
add_test(NAME preview COMMAND /usr/lib/qt6/bin/qmltestrunner -input ${CMAKE_CURRENT_SOURCE_DIR}/tests/tst_preview.qml)
set_tests_properties(preview PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_library(omtbridge MODULE src/focus_bridge.cpp)
target_include_directories(omtbridge PRIVATE src)
target_link_libraries(omtbridge PRIVATE Fcitx5::Core Fcitx5::Module::DBus Qt6::Core)
add_executable(test_bridge tests/test_bridge.cpp)
target_link_libraries(test_bridge PRIVATE Fcitx5::Core Fcitx5::Module::DBus Qt6::Core Qt6::DBus)
add_test(NAME bridge COMMAND dbus-run-session -- $<TARGET_FILE:test_bridge> --disable all --enable dbus,keyboard)
configure_file(omtbridge.conf.in omtbridge.conf @ONLY)
add_executable(omt-backend src/backend.cpp src/focus_link.cpp src/focus_link.h)
target_link_libraries(omt-backend PRIVATE omt_engine Qt6::Core Qt6::DBus)

add_test(NAME touchkey COMMAND /usr/lib/qt6/bin/qmltestrunner -input ${CMAKE_CURRENT_SOURCE_DIR}/tests/tst_touchkey.qml)
set_tests_properties(touchkey PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
set(BUILD_TESTING ON)
add_subdirectory(vendor/handwriting)
add_executable(omt-handwriting src/handwriting_worker.cpp)
target_link_libraries(omt-handwriting PRIVATE MetasequoiaHandwriting Qt6::Core)

add_executable(test_learning tests/test_learning.cpp)
target_link_libraries(test_learning PRIVATE omt_engine)
add_test(NAME learning COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_learning.py)
