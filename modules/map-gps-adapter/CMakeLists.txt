cmake_minimum_required(VERSION 3.21)
project(workbench_map_gps VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Positioning Test)
add_library(gps_source STATIC source.cpp source.h)
set_target_properties(gps_source PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(gps_source PUBLIC Qt6::Core Qt6::Network Qt6::Positioning)
add_library(workbench_gpsd MODULE factory.cpp metadata.json)
target_link_libraries(workbench_gpsd PRIVATE gps_source)
set_target_properties(workbench_gpsd PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/position")
add_executable(test_gps tests.cpp)
target_link_libraries(test_gps PRIVATE gps_source Qt6::Test)
enable_testing()
add_test(NAME gps_source COMMAND test_gps)
add_executable(factory_probe factory_probe.cpp)
target_link_libraries(factory_probe PRIVATE Qt6::Core Qt6::Positioning)
add_test(NAME plugin_factory COMMAND factory_probe $<TARGET_FILE:workbench_gpsd>)
set_tests_properties(plugin_factory PROPERTIES ENVIRONMENT "QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins")
